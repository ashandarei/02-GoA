cmake_minimum_required(VERSION 3.22)
project(ZeroTwoGoATests LANGUAGES CXX)
if(NOT WIN32)
    message(FATAL_ERROR "Build Windows x64 tests; run the resulting EXE on Windows or under Wine/Proton.")
endif()
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(02_goa_tests test_hardening.cpp
    ../src/shared/kh02/kh02_owner_pipe.cpp
    ../src/shared/kh02/kh02_memory_signature.cpp)
target_include_directories(02_goa_tests PRIVATE ../src)
target_link_libraries(02_goa_tests PRIVATE bcrypt advapi32 shell32)
enable_testing()
add_test(NAME companion_hardening COMMAND 02_goa_tests)
set_tests_properties(companion_hardening PROPERTIES TIMEOUT 30)

# Companion-only regression tests.
find_package(Git REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/kh3-json-dependency.cmake")
add_executable(kh02_asset_index_test kh02_asset_index_test.cpp)
target_include_directories(kh02_asset_index_test PRIVATE ../src)
target_link_libraries(kh02_asset_index_test PRIVATE kh3ap_json_headers)
if(MSVC)
    target_compile_options(kh02_asset_index_test PRIVATE /UNDEBUG)
else()
    target_compile_options(kh02_asset_index_test PRIVATE -UNDEBUG)
endif()
add_test(NAME kh02_asset_index COMMAND kh02_asset_index_test)
add_executable(test_kh02_movie_frames test_kh02_movie_frames.cpp ../src/probe/movie_frames.cpp)
add_executable(test_kh02_pak_read_profile test_kh02_pak_read_profile.cpp ../src/probe/pak_read_profile.cpp)
target_link_libraries(test_kh02_pak_read_profile PRIVATE kh3ap_json_headers)
add_executable(test_kh02_parallel_pak_read test_kh02_parallel_pak_read.cpp)
foreach(_test IN ITEMS test_kh02_movie_frames test_kh02_pak_read_profile test_kh02_parallel_pak_read)
    target_include_directories(${_test} PRIVATE ../src ../src/support)
    if(MSVC)
        target_compile_options(${_test} PRIVATE /UNDEBUG)
    else()
        target_compile_options(${_test} PRIVATE -UNDEBUG)
    endif()
    add_test(NAME ${_test} COMMAND ${_test})
    set_tests_properties(${_test} PROPERTIES TIMEOUT 30)
endforeach()
